# claims-machine.txt — atomic claims from the LCEIC report's Findings

# One atomic claim per line, decomposed from the report's 55 Findings (F1–F55). Recommendations
# (R1–R11) ask the government to do something rather than state a fact, so they cannot be true or
# false and are excluded here — they live in report-summary.md.
#
# This is the CLAIM side of a faithfulness assay. The SOURCE side is sources/ (hearing transcripts)
# and the cited submissions. Each line's src= field names the witness/submission the report's body
# rests the claim on, so a faithfulness check can go to the truth-maker.
#
# Line format (pipe-delimited, 5 fields):
#   <id> | route=<r> | ref=§<section> p<printed-page> | <claim> | src=<witness/submission + tx page + [fn]>
#
# route= (aligns with the axis boundary in CLAUDE.md — what can settle each claim):
#   evidence   — factual claim about the world; needs an external truth-maker (web/dataset grounding).
#   source     — compresses witness/submission testimony; check against sources/ (faithfulness).
#   evaluative — Committee value judgment / opinion ("disappointing", "should", "never harder");
#                only checkable as "did stakeholders say this", never groundable as world-fact.
#   data-gap   — claim about a dataset's own limits (a breakdown "is not released").
#
# tx = "Transcript of evidence"; date = hearing day → sources/hearings/<date>/. [fnN] = report footnote.
